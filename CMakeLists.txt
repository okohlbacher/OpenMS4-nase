cmake_minimum_required(VERSION 3.24)
project(OpenMSNASE VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
endif()
include(cmake/OpenMS4Dependencies.cmake)
openms4_find_core()
openms4_find_package(OpenMSCLI)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<0:>")
include(cmake/OpenMS4Tools.cmake)
openms4_load_tools(package_tools)
foreach(tool IN LISTS package_tools)
  add_executable(${tool} src/${tool}.cpp)
  target_link_libraries(${tool} PRIVATE OpenMS::CLI OpenMS::Core)
endforeach()
openms4_install_tools(nase bin ${package_tools})
option(OPENMS4_REGRESSION_TESTS "Locate fixtures for the installed-tool regression harness" OFF)
if(OPENMS4_REGRESSION_TESTS)
  openms4_find_package(OpenMSTestData)
  message(STATUS "Run the installed-tools harness for TOPP_NucleicAcidSearchEngine tests and output comparisons")
endif()
